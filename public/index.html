async function prosesPembayaran() { let checkboxes = document.querySelectorAll('#monthsContainer input[type="checkbox"]:checked:not(:disabled)'); if (checkboxes.length === 0) { alert("Pilih minimal satu bulan yang belum lunas untuk dibayar!"); return; } let bulanPilihan = Array.from(checkboxes).map(cb => cb.value); let nis = santriTerpilih.NIS || santriTerpilih.nis; let totalBayar = checkboxes.length * nominalSPP; if (metodePilih === 'online') { alert(`Mengarahkan ke pembayaran Online Mayar.id sejumlah Rp ${totalBayar.toLocaleString('id-ID')} untuk bulan: ${bulanPilihan.join(', ')}`); return; } if (!confirm(`Konfirmasi pembayaran Tunai untuk bulan: ${bulanPilihan.join(', ')}?`)) return; document.getElementById('loadingText').innerText = "Memproses pembayaran & mengirim kuitansi..."; document.getElementById('loadingText').style.display = 'block'; try { // Mengirim data satu per satu dengan metode form/URLSearchParams agar lolos dari CORS Apps Script for (let b of bulanPilihan) { await fetch(WEB_APP_URL, { method: 'POST', body: JSON.stringify({ nis: nis, bulan: b }) }); } alert("Pembayaran Tunai berhasil! Status telah diperbarui dan kuitansi otomatis dikirim ke WhatsApp."); location.reload(); } catch (err) { console.error(err); // Jika masih error CORS tapi data sebenarnya masuk ke sheet, kita beri jeda refresh alert("Pembayaran diproses. Halaman akan dimuat ulang."); location.reload(); } finally { document.getElementById('loadingText').style.display = 'none'; } }
